cmake_minimum_required(VERSION 3.16)
project(unitree_legged_real)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)

find_package(unitree_go REQUIRED)
find_package(unitree_rl_controller REQUIRED)
find_package(yaml-cpp REQUIRED)
if(DEFINED ENV{Torch_DIR} AND NOT Torch_DIR)
  set(Torch_DIR "$ENV{Torch_DIR}")
endif()
find_package(Torch REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
# SDK2 is confined to a separate SDK-only executable.
set(BUILD_EXAMPLES OFF CACHE BOOL "SDK examples" FORCE)
add_subdirectory(library/unitree_sdk2 EXCLUDE_FROM_ALL)
add_library(go2_motion_mode STATIC src/go2_motion_mode.cpp)
target_include_directories(go2_motion_mode PUBLIC include)
target_link_libraries(go2_motion_mode unitree_sdk2)
add_executable(go2_mode_switch src/go2_mode_switch.cpp)
target_link_libraries(go2_mode_switch go2_motion_mode)
# Vendored CycloneDDS-CXX headers require C++17 with GCC 13 (template destructors).
# These SDK-only targets do not consume Torch's C++20 headers.
set_target_properties(go2_motion_mode go2_mode_switch PROPERTIES CXX_STANDARD 17)
# DT_RUNPATH loses to Jazzy's LD_LIBRARY_PATH and mixes SDK C++ DDS with ROS C DDS.
# DT_RPATH has priority and applies to transitive SDK dependencies as well.
# Keep this setting confined to the SDK-only executable.
target_link_options(go2_mode_switch PRIVATE "LINKER:--disable-new-dtags")
set_target_properties(go2_mode_switch PROPERTIES INSTALL_RPATH "$ORIGIN/sdk2")
# Private DDS directory for the SDK-only process; never export it to ROS.
install(DIRECTORY "library/unitree_sdk2/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}/"
  DESTINATION lib/${PROJECT_NAME}/sdk2 FILES_MATCHING PATTERN "*.so*")
# R2 pure IO/CRC/safety library: message types only, no Unitree DDS libraries.
set(RARS_ARM_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../rars_arm_sdk" CACHE PATH "Explicit RARS SDK source")
set(RARS_ARM_BUILD_QT_EXAMPLE OFF CACHE BOOL "No hardware GUI" FORCE)
set(RARS_ARM_BUILD_PYTHON OFF CACHE BOOL "No Python module" FORCE)
add_subdirectory("${RARS_ARM_SDK_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/rars_sdk" EXCLUDE_FROM_ALL)
add_library(real_safety_io SHARED src/safety_io.cpp src/rars_bridge.cpp)
target_include_directories(real_safety_io PUBLIC include "${RARS_ARM_SDK_ROOT}/include" PRIVATE library/unitree_sdk2/example/state_machine)
ament_target_dependencies(real_safety_io unitree_go)
add_library(rars_sdk_readonly STATIC src/rars_sdk_readonly.cpp)
target_link_libraries(rars_sdk_readonly PUBLIC real_safety_io rars_arm::sdk)
install(TARGETS real_safety_io LIBRARY DESTINATION lib)
add_library(real_controller_core SHARED src/real_controller_core.cpp src/navigation_command_adapter.cpp)
target_include_directories(real_controller_core PUBLIC include ${TORCH_INCLUDE_DIRS})
ament_target_dependencies(real_controller_core unitree_rl_controller unitree_go)
target_link_libraries(real_controller_core real_safety_io ${TORCH_LIBRARIES} yaml-cpp)
add_library(real_r3_commissioning SHARED src/r3_commissioning.cpp)
target_include_directories(real_r3_commissioning PUBLIC include)
target_link_libraries(real_r3_commissioning PUBLIC real_safety_io yaml-cpp)
add_executable(go2_r3_commissioning src/go2_r3_commissioning.cpp)
target_link_libraries(go2_r3_commissioning real_r3_commissioning real_controller_core)
ament_target_dependencies(go2_r3_commissioning rclcpp std_msgs std_srvs geometry_msgs unitree_go unitree_rl_controller)
add_library(rars_auto_home STATIC src/rars_auto_home.cpp)
target_include_directories(rars_auto_home PUBLIC include "${RARS_ARM_SDK_ROOT}/include")
add_executable(rars_r3_owner src/rars_r3_owner.cpp)
target_link_libraries(rars_r3_owner rars_sdk_readonly rars_auto_home yaml-cpp)
ament_target_dependencies(rars_r3_owner rclcpp std_msgs std_srvs)
add_executable(r3_policy_timing tests/r3_policy_timing.cpp)
target_link_libraries(r3_policy_timing real_controller_core)
ament_target_dependencies(r3_policy_timing unitree_rl_controller)
add_executable(ros2_rl_go2 src/ros2_rl_go2.cpp)
target_link_libraries(ros2_rl_go2 real_controller_core ${TORCH_LIBRARIES} yaml-cpp)
ament_target_dependencies(ros2_rl_go2 rclcpp std_msgs sensor_msgs geometry_msgs unitree_go unitree_rl_controller)
foreach(target real_controller_core ros2_rl_go2 go2_r3_commissioning r3_policy_timing)
  set_target_properties(${target} PROPERTIES BUILD_RPATH "${TORCH_INSTALL_PREFIX}/lib" INSTALL_RPATH "${TORCH_INSTALL_PREFIX}/lib")
endforeach()

option(BUILD_LEGACY_SIM "Build retained simulation target" OFF)

  find_library(LCM_LIBRARY lcm REQUIRED)
  if(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
    set(LEGACY_ARCH arm64)
  elseif(CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
    set(LEGACY_ARCH amd64)
